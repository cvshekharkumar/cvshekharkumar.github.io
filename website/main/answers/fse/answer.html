            dayOut.textContent = "Enter 1 to 7";
        } else {
            dayOut.textContent = "Day " + n + " is " + days[n - 1];
        }

    });
}

function initApp() {
    var ratingInput = document.getElementById('ratingInput');
    var showBtn = document.getElementById('showBtn');
    var stars = document.getElementById('stars');

    showBtn.addEventListener('click', function () {
        var n = parseFloat(ratingInput.value);
        if (isNaN(n) || n < 0 || n > 5 || Math.floor(n) !== n) {
            stars.textContent = "Enter 0 to 5";
        } else {
            var str = "";
            for (var i = 0; i < n; i++) {
                str += "\u2605";
            }
            for (var j = 0; j < 5 - n; j++) {
                str += "\u2606";
            }
            stars.textContent = str;
        }

    });
}

if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initApp);
} else {
    initApp();
}

function initApp() {
    var sentenceInput = document.getElementById('sentenceInput');
    var reverseBtn = document.getElementById('reverseBtn');
    var reversedOut = document.getElementById('reversedOut');

    reverseBtn.addEventListener('click', function () {
        var text = sentenceInput.value.trim();
        if (text === "") {
            reversedOut.textContent = "Enter a sentence";
        } else {
            var raw = text.split(" ");
            var words = [];
            for (var i = 0; i < raw.length; i++) {
                if (raw[i] !== "") {
                    words.push(raw[i]);
                }
            }
            var reversed = [];
            for (var j = words.length - 1; j >= 0; j--) {
                reversed.push(words[j]);
            }
            reversedOut.textContent = reversed.join(" ");
        }

    });
}
